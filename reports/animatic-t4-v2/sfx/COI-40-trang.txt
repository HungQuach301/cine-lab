(function () {
  var CLE = 'laso_popup_soutien_v1';
  // En aperçu admin : mémorisé le temps de l'onglet seulement, pour pouvoir retester
  var stockage = 'localStorage';
  var popup = document.getElementById('popup-soutien');
  if (!popup) return;

  var force = /[?&]popup-soutien\b/.test(location.search);
  var dejaVu = false;
  try { dejaVu = window[stockage].getItem(CLE) === '1'; } catch (e) {}
  if (dejaVu && !force) return;

  function memoriser() { try { window[stockage].setItem(CLE, '1'); } catch (e) {} }

  function fermer() {
    memoriser();
    popup.classList.remove('ps-visible');
    document.removeEventListener('keydown', surTouche);
    setTimeout(function () { popup.hidden = true; }, 300);
  }

  function surTouche(e) { if (e.key === 'Escape') fermer(); }

  popup.addEventListener('click', function (e) {
    if (e.target === popup || e.target.closest('[data-ps-fermer]')) fermer();
    else if (e.target.closest('[data-ps-vu]')) memoriser();
  });

  setTimeout(function () {
    popup.hidden = false;
    requestAnimationFrame(function () { popup.classList.add('ps-visible'); });

    // Focus sur la fenêtre (et non sur le 1er choix, qui paraîtrait présélectionné) ; Tab mène aux choix
    popup.querySelector('.ps-boite').focus({ preventScroll: true });
  }, 1200);
})();
